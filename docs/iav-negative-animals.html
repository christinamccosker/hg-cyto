9 IAV Negative Animals

iavneg_plsda <- mixOmics::plsda(datascale_iavneg[,2:10],
                                datascale_iavneg$iavser,
                                scale=FALSE,
                                ncomp=9)

iavneg_plsda_data <- 
  data.frame(x = iavneg_plsda$variates$X[,1], 
             y = iavneg_plsda$variates$X[,2],
             iav = iavneg_plsda$Y)

# Set x and y axis labels based on prop of variation explained
labelx_iavneg <- paste("LV1 (", 
                round(abs(iavneg_plsda$prop_expl_var$X[1]*100), digits=0),
                "%)",
                sep="")

labely_iavneg <- paste("LV2 (", 
                round(abs(iavneg_plsda$prop_expl_var$X[2]*100), digits=0),
                "%)",
                sep="")

iavneg_plsda_ggplot <- 
  ggplot(iavneg_plsda_data, 
         aes(x=x, y=y, col=iav)) +
  geom_point() +
  stat_ellipse() +
  labs(x=labelx_iavneg,
       y =labely_iavneg) +
  scale_color_manual("C. IAV Serology Status", 
                     values=c("#9d9d9d", "black")) +
  theme_bw() +
  theme(panel.grid=element_blank(),
        legend.position=c(0.14, 0.13))
