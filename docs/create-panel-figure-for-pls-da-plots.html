11 Create panel figure for PLS-DA plots

11.1 Libraries

library(mixOmics)
library(tidyverse)
library(gridExtra)
library(reshape2)

11.2 Data

cyto <- 
  read.csv("Output Files/cleaned_cytokine.csv")

cytoscale <- 
  read.csv("Output Files/cleaned_cytokine_scaled.csv")

meta <- 
  read.csv("Output Files/metadata_cyto.csv") %>% 
  mutate(iav=gsub("neg", "IAV-", iav),
         iav=gsub("pos", "IAV+", iav))

# Merge cytokine & metadata 
data <- merge(cyto, meta, by="Sample")
datascale <- merge(cytoscale, meta, by="Sample")

# Prepare data
data_molt <-
  data %>%
  filter(!is.na(molt.stage))

datascale_molt <- 
  datascale %>% 
  filter(!is.na(molt.stage))

data_bc <-
  data %>%
  filter(!is.na(bc_bin))

datascale_bc <-
  datascale %>%
  filter(!is.na(bc_bin))


data_iavneg <-
  data %>% 
  filter(iav=="IAV-")

datascale_iavneg <-
  datascale %>% 
  filter(iav=="IAV-")

11.3 Plot!

11.3.1 Molt Stage

molt_plsda <- mixOmics::plsda(datascale_molt[,2:10],
                              datascale_molt$molt.stage,
                              scale=FALSE,
                              ncomp=9)
molt_plsda_data <- 
  data.frame(x = molt_plsda$variates$X[,1], 
             y = molt_plsda$variates$X[,2],
             molt.stage = molt_plsda$Y)

# Set x and y axis labels based on prop of variation explained
labelx_molt <- paste("LV1 (", 
                round(abs(molt_plsda$prop_expl_var$X[1]*100), digits=0),
                "%)",
                sep="")

labely_molt <- paste("LV2 (", 
                round(abs(molt_plsda$prop_expl_var$X[2]*100), digits=0),
                "%)",
                sep="")


molt_plsda_ggplot <- 
  ggplot(molt_plsda_data, 
         aes(x=x, y=y, col=molt.stage)) +
  geom_point() +
  stat_ellipse() +
  labs(x=labelx_molt,
       y =labely_molt) +
  scale_color_manual("A. Molt Stage", 
                     values=c("#9d9d9d","#008ba2", "black")) +
  theme_bw() +
  theme(panel.grid=element_blank(),
        legend.position=c(0.85, 0.17))
